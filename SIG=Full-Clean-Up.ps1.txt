# ============================================================# SIG Full Cleanup — Consolidate Core UI Files into C:\SIG\ui# Author: Mark J. Latsha + Microsoft Copilot# ============================================================Write-Host "Starting SIG full cleanup..." -ForegroundColor Cyan# 1. Ensure ui folder exists$uiPath = "C:\SIG\ui"if (!(Test-Path $uiPath)) {    Write-Host "ERROR: $uiPath does not exist. Aborting." -ForegroundColor Red    return}# 2. Core SIG files to consolidate$files = @(    "intro_panel.py",    "intro_panel_widget.py",    "main_window.py",    "artifact_panel.py",    "animation_engine.py",    "cinematic_mode.py",    "cinematic_mode_v2.py")foreach ($name in $files) {    Write-Host "`n=== Processing: $name ===" -ForegroundColor Yellow    $matches = Get-ChildItem -Path "C:\SIG" -Recurse -Filter $name -ErrorAction SilentlyContinue    if ($matches.Count -eq 0) {        Write-Host "No copies found. Skipping." -ForegroundColor DarkGray        continue    }    Write-Host "Found:" -ForegroundColor Green    $matches | ForEach-Object { Write-Host " - $($_.FullName)" }    $dest = Join-Path $uiPath $name    # 3. Choose authoritative version:    #    - If a copy already exists in ui/, use that.    #    - Otherwise, use the newest file.    $uiCopy = $matches | Where-Object { $_.DirectoryName -eq $uiPath } | Select-Object -First 1    if ($uiCopy) {        $authoritative = $uiCopy        Write-Host "Using existing ui/ version as authoritative:" -ForegroundColor Cyan        Write-Host " -> $($authoritative.FullName)"    } else {        $authoritative = $matches | Sort-Object LastWriteTime -Descending | Select-Object -First 1        Write-Host "Using newest file as authoritative:" -ForegroundColor Cyan        Write-Host " -> $($authoritative.FullName)"        Copy-Item -Path $authoritative.FullName -Destination $dest -Force        Write-Host "Copied authoritative version to ui/: $dest" -ForegroundColor Green    }    # 4. Remove all non-ui duplicates    $duplicates = $matches | Where-Object { $_.FullName -ne $authoritative.FullName -and $_.DirectoryName -ne $uiPath }    if ($duplicates.Count -gt 0) {        Write-Host "Removing non-ui duplicates..." -ForegroundColor Yellow        foreach ($dup in $duplicates) {            Write-Host " - Removing: $($dup.FullName)"            Remove-Item $dup.FullName -Force        }    } else {        Write-Host "No non-ui duplicates to remove." -ForegroundColor DarkGray    }}Write-Host "`nSIG cleanup complete. Core files consolidated in C:\SIG\ui." -ForegroundColor Green
